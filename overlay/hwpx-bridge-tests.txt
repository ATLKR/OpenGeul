  function hwpxSaveSetup() {
    const bridge = new TauriBridge();
    const handle = writeHandle();
    fsOpenMock.mockResolvedValue(handle);
    applyOpenResult(bridge, nativeOpenResult({ fileName: 'source.hwpx', sourcePath: '/tmp/source.hwpx', format: 'hwpx', dirty: true }));
    invokeMock.mockImplementation(async (command: string, args: Record<string, unknown>) => {
      if (command === 'check_external_modification') return { changed: false };
      if (command.startsWith('prepare_staged_')) return `${args.targetPath}.hop-save-0123456789abcdef0123456789abcdef.tmp`;
      if (command.startsWith('commit_staged_')) return { docId: 'doc-1', sourcePath: args.targetPath, format: command.includes('hwpx') ? 'hwpx' : 'hwp', revision: 2, dirty: false, warnings: [] };
      if (command === 'note_finder_recent_document' || command === 'mark_document_dirty') return undefined;
      throw new Error(`unexpected command ${command}`);
    });
    return { bridge, handle };
  }

  function replacementGate() {
    let resolve!: () => void;
    const promise = new Promise<void>((ready) => { resolve = ready; });
    return { promise, resolve };
  }

  async function replacementSetup() {
    const bridge = new TauriBridge();
    const blocked = replacementGate();
    const entered = replacementGate();
    let blockRead = false;
    let blockNative: string | null = null;
    let failNewCleanup = false;
    let content = '';
    getWasmMock(bridge, 'loadDocumentMock').mockImplementation((bytes: Uint8Array) => {
      content = new TextDecoder().decode(bytes);
      return { pageCount: 1, fontsUsed: [] };
    });
    getWasmMock(bridge, 'createNewDocumentMock').mockImplementation(() => {
      content = 'BLANK';
      return { pageCount: 1, fontsUsed: [] };
    });
    fsOpenMock.mockImplementation(async (path: string) => {
      const handle = readHandle(path === '/tmp/old.hwp' ? [79, 76, 68] : [78, 69, 87]);
      const read = handle.read.getMockImplementation()!;
      handle.read.mockImplementation(async (bytes: Uint8Array) => {
        if (blockRead && path === '/tmp/new.hwp') {
          entered.resolve();
          await blocked.promise;
        }
        return read(bytes);
      });
      return handle;
    });
    messageMock.mockResolvedValue('저장 안 함');
    invokeMock.mockImplementation(async (command: string, args: Record<string, unknown>) => {
      if (command === blockNative) {
        entered.resolve();
        await blocked.promise;
      }
      if (command === 'open_document_tracking') {
        return args.path === '/tmp/old.hwp'
          ? nativeOpenResult({ docId: 'old-doc', fileName: 'old.hwp', sourcePath: '/tmp/old.hwp' })
          : nativeOpenResult({ docId: 'incoming-doc', fileName: 'new.hwp', sourcePath: '/tmp/new.hwp' });
      }
      if (command === 'create_document') {
        return nativeOpenResult({ docId: 'incoming-doc', fileName: '새 문서.hwp', sourcePath: null });
      }
      if (command === 'close_document' && args.docId === 'incoming-doc' && failNewCleanup) {
        throw new Error('native cleanup failed');
      }
      if (['prepare_document_open', 'mark_document_dirty', 'note_finder_recent_document',
        'record_recent_document', 'close_document'].includes(command)) return undefined;
      throw new Error(`unexpected command ${command}`);
    });
    await bridge.openDocumentByPath('/tmp/old.hwp');
    expect(content).toBe('OLD');
    expect(bridge.hasUnsavedChanges()).toBe(false);
    // Keep the native transport controlled across the deferred response. The
    // actual public bridge operations, chunked reads and WASM replacement run.
    const transport = bridge as unknown as {
      invoke(command: string, args?: Record<string, unknown>): Promise<unknown>;
    };
    vi.spyOn(transport, 'invoke').mockImplementation((command, args = {}) =>
      Promise.resolve(invokeMock(command, args)));
    invokeMock.mockClear();
    getWasmMock(bridge, 'loadDocumentMock').mockClear();
    return {
      bridge, blocked, entered,
      content: () => content,
      edit: () => { content += ' +EDIT'; bridge.markDocumentDirty(); },
      deferRead: () => { blockRead = true; },
      deferNative: (command: string) => { blockNative = command; },
      rejectNewCleanup: () => { failNewCleanup = true; },
      deferPassword: () => {
        getWasmMock(bridge, 'loadDocumentMock').mockImplementation(() => {
          throw new Error('비밀번호가 필요한 암호 문서');
        });
        const wasm = bridge as unknown as {
          loadDocumentWithPassword(bytes: Uint8Array, password: string, fileName: string): void;
          getDocumentInfo(): { pageCount: number; fontsUsed: string[] };
        };
        const passwordLoad = vi.fn((bytes: Uint8Array) => { content = new TextDecoder().decode(bytes); });
        wasm.loadDocumentWithPassword = passwordLoad;
        wasm.getDocumentInfo = () => ({ pageCount: 1, fontsUsed: [] });
        passwordPrompt.mockImplementation(async () => {
          entered.resolve();
          await blocked.promise;
          return 'correct';
        });
        return passwordLoad;
      },
    };
  }

  function expectEditedDocumentPreserved(
    setup: Awaited<ReturnType<typeof replacementSetup>>, content: string, encryptedAttempt = false,
  ) {
    expect(setup.content()).toBe(content);
    expect(setup.bridge.hasUnsavedChanges()).toBe(true);
    const native = setup.bridge as unknown as { docId: string; sourcePath: string | null; revision: number };
    expect(native.docId).toBe('old-doc');
    expect(native.sourcePath).toBe('/tmp/old.hwp');
    expect(native.revision).toBe(1);
    expect(document.title).toBe('• old.hwp - HOP');
    expect(invokeMock).not.toHaveBeenCalledWith('close_document', { docId: 'old-doc' });
    expect(invokeMock).toHaveBeenCalledWith('close_document', { docId: 'incoming-doc' });
    expect(getWasmMock(setup.bridge, 'loadDocumentMock')).toHaveBeenCalledTimes(encryptedAttempt ? 1 : 0);
    expect(getWasmMock(setup.bridge, 'createNewDocumentMock')).not.toHaveBeenCalled();
    expect(invokeMock).not.toHaveBeenCalledWith('record_recent_document', { path: '/tmp/new.hwp' });
  }

  it.each([false, true])('preserves edits during a replacement read after initial dirty=%s consent', async (initialDirty) => {
    const setup = await replacementSetup();
    if (initialDirty) setup.edit();
    setup.deferRead();
    const opening = setup.bridge.openDocumentByPath('/tmp/new.hwp');
    await setup.entered.promise;
    setup.edit();
    const edited = setup.content();
    setup.blocked.resolve();
    await expect(opening).rejects.toThrow('문서 교체를 준비하는 동안 현재 문서가 변경');
    expectEditedDocumentPreserved(setup, edited);
    expect(messageMock).toHaveBeenCalledTimes(initialDirty ? 1 : 0);
  });
